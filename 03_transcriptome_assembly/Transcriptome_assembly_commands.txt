# Workflow
# Step 1: Download SRA data
prefetch SRRXXXXXX
fastq-dump --split-files SRRXXXXXX

# Step 2: Quality control and preprocessing
fastqc SRRXXXXXX_1.fastq SRRXXXXXX_2.fastq
trimmomatic PE SRRXXXXXX_1.fastq SRRXXXXXX_2.fastq \
              SRRXXXXXX_1_trimmed.fastq SRRXXXXXX_1_unpaired.fastq \
              SRRXXXXXX_2_trimmed.fastq SRRXXXXXX_2_unpaired.fastq \
              ILLUMINACLIP:TruSeq3-PE.fa:2:30:10 \
              LEADING:3 TRAILING:3 SLIDINGWINDOW:4:15 MINLEN:36

# Step 3: Assemble the transcriptome
Trinity --seqType fq --max_memory 50G --left SRRXXXXXX_1_trimmed.fastq --right SRRXXXXXX_2_trimmed.fastq --CPU 10 --output trinity_out_dir

# Step 4: Post-assembly processing
busco -i trinity_out_dir/Trinity.fasta -l eukaryota_odb9 -o busco_out -m transcriptome
TransDecoder.LongOrfs -t trinity_out_dir/Trinity.fasta
TransDecoder.Predict -t trinity_out_dir/Trinity.fasta

# Step 5: Functional annotation
